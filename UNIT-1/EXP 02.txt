import torch
import transformers
from transformers import pipeline

print("=" * 55)
print("HUGGING FACE TRANSFORMER MODEL")
print("=" * 55)


print("\nPython libraries loaded successfully!")
print("Transformers version:", transformers.__version__)
print("PyTorch version:", torch.__version__)

device = 0 if torch.cuda.is_available() else -1

if device == 0:
    print("Computing device: GPU")
else:
    print("Computing device: CPU")

model_name = (
    "distilbert/distilbert-base-uncased-finetuned-sst-2-english"
)

print("\nLoading pre-trained model...")
sentiment_pipeline = pipeline(
    task="sentiment-analysis",
    model=model_name,
    device=device
)

print("Model loaded successfully!")


texts = [
    "I love learning Artificial Intelligence!",
    "This movie is very disappointing.",
    "Hugging Face makes NLP easy to use.",
    "The food was delicious and excellent.",
    "The service was terrible and slow."
]


print("\n" + "=" * 55)
print("SENTIMENT ANALYSIS RESULTS")
print("=" * 55)

results = sentiment_pipeline(texts)

for i, (text, result) in enumerate(zip(texts, results), start=1):
    print(f"\nSentence {i}: {text}")
    print("Sentiment :", result["label"])
    print("Confidence:", round(result["score"] * 100, 2), "%")


print("\n" + "=" * 55)
print("CUSTOM TEXT ANALYSIS")
print("=" * 55)

custom_text = input("Enter a sentence: ")

custom_result = sentiment_pipeline(custom_text)[0]

print("\nInput text:", custom_text)
print("Predicted sentiment:", custom_result["label"])
print(
    "Confidence:",
    round(custom_result["score"] * 100, 2),
    "%"
)


print("\n" + "=" * 55)
print("EXPERIMENT COMPLETED SUCCESSFULLY!")
print("Hugging Face model loaded and tested.")
print("=" * 55)
